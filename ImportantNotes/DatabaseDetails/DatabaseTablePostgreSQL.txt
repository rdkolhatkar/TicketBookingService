-- ------------------------------------------------------------------------
-- Create PostgreSQL schema in Public database
-- ------------------------------------------------------------------------
CREATE SCHEMA IF NOT EXISTS ticketbookingapi;
-- Set the schema for your session
SET search_path TO ticketbookingapi;
-- ------------------------------------------------------------------------
-- Create registration table
-- ------------------------------------------------------------------------
CREATE TABLE ticketbookingapi.registration (
    user_id CHAR(36) PRIMARY KEY,   -- UUID string
    user_name VARCHAR(255) UNIQUE NOT NULL,
    email VARCHAR(255) NOT NULL,
    first_name VARCHAR(255) NOT NULL,
    last_name VARCHAR(255) NOT NULL,
    mobile VARCHAR(10) NOT NULL,
    password VARCHAR(255) NOT NULL
);

-- View all users
SELECT * FROM registration;

-- ------------------------------------------------------------------------
-- Create bookings table
-- ------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS bookings (
    booking_id BIGSERIAL PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL,
    booked_by_name VARCHAR(255) NOT NULL,
    movie_name VARCHAR(255) NOT NULL,
    number_of_tickets INT NOT NULL
);

-- View all bookings
SELECT * FROM bookings;
-- Make user_id FOREIGN KEY
ALTER TABLE bookings
ADD CONSTRAINT fk_user
FOREIGN KEY (user_id) REFERENCES registration(user_id);

-- Delete Table
DROP TABLE IF EXISTS ticketbookingapi.registration;
